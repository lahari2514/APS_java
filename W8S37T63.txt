
import java.util.*;

class Solution {
    public int[] shortestAlternatingPaths(int n, int[][] redEdges,
                                          int[][] blueEdges) {

        // graph[0] = red edges
        // graph[1] = blue edges
        List<List<Integer>>[] graph = new ArrayList[2];

        graph[0] = new ArrayList<>();
        graph[1] = new ArrayList<>();

        for (int i = 0; i < n; i++) {
            graph[0].add(new ArrayList<>());
            graph[1].add(new ArrayList<>());
        }

        // Add red edges
        for (int[] edge : redEdges) {
            graph[0].get(edge[0]).add(edge[1]);
        }

        // Add blue edges
        for (int[] edge : blueEdges) {
            graph[1].get(edge[0]).add(edge[1]);
        }

        // answer[i] = shortest distance to node i
        int[] answer = new int[n];
        Arrays.fill(answer, -1);

        // color 0 = red was used last
        // color 1 = blue was used last

        // Start from node 0.
        // We can take either red or blue as the first edge.
        queue.offer(new int[]{0, 0});
        queue.offer(new int[]{0, 1});

        visited[0][0] = true;
        visited[0][1] = true;

        answer[0] = 0;
