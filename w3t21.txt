import java.util.*;

class StockSpanner {

    // Stack stores: [price, span]
    Stack<int[]> stack;

    public StockSpanner() {
        stack = new Stack<>();
    }

    public int next(int price) {
        
        int span = 1;

        // Combine spans of all previous prices <= current price
        while (!stack.isEmpty() && stack.peek()[0] <= price) {
            span += stack.pop()[1];
        }

        // Store current price and its total span
        stack.push(new int[]{price, span});

        return span;
    }
}
